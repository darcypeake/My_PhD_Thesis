\chapter{Symbolic coefficient validation}
\label{ap:symb-val}

This appendix documents the \textsc{Mathematica} validation of the
\textsc{ARES} coefficients $h_{22}$ and $h_{21}$ against the SCET
subtraction coefficients $\mathcal{C}^{(2)}_1$ and $\mathcal{C}^{(2)}_0$,
discussed in Sec.~\ref{subsec:NNLL-coeff}. The complete notebook is available
at Ref.~\cite{peake2026validation}.\footnote{The coefficients as printed in
Ref.~\cite{Gaunt:2015pea} omit the final four lines of both
Eqs.~\eqref{eq:C21} and~\eqref{eq:C20}, which appears to be a misprint since
these terms are required for the coefficients to be complete.}

Within this notebook, the ingredients of Sec.~\ref{ap:subtract-ingred} and
the \textsc{ARES} coefficients of Eqs.~\eqref{eq:h11}--\eqref{eq:h21} are
entered directly. Uppercase symbols denote the full expansion of a
coefficient, following Eq.~\eqref{eq:subt-mn-exp}, so that for example
\texttt{J[1,0]} corresponds to
$J^{(1)}_{\kappa_3,0}(\lambda_3)=J^{(1)}_{\kappa_3,0}
+J^{(1)}_{\kappa_3,1}\ln\lambda_3$. The remaining correspondence with the
notation of this thesis is given in Table~\ref{tab:mma-symbols}.

\section{Validation of \texorpdfstring{$h_{22}$}{h22}}
\label{ap:mathematica-h22}

We begin by validating $\mathcal{C}^{(2)}_1$, given by
\begin{equation}
\label{eq:C21}
\begin{aligned}
&\mathcal{C}^{(2)}_1=\frac{B_{\kappa_1,1}^{(2)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(2)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+
    J_{\kappa_3,1}^{(2)}(\lambda_3)
    +\widehat{S}^{(2)}_{\kappa,1}\left(\lambda_s\right)\\[0.5ex]
    &\qquad + H^{(1)}_{\rm SCET}\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+
    J_{\kappa_3,1}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,1}\!\left(\{\hat{p}_\ell\},\lambda_s\right)\right)\\[0.5ex]
    &\qquad +2\left[\left(\frac{B^{(1)}_{\kappa_1,0}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B^{(1)}_{\kappa_2,0}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right)\left(J_{\kappa_1,0}^{(1)}(\lambda_3)+\widehat{S}_{\kappa,0}^{(1)}(\lambda_s)\right)+\frac{B^{(1)}_{\kappa_1,0}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\frac{B^{(1)}_{\kappa_2,0}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right]\\[0.5ex]
    &\qquad -\frac{\pi^2}{3}\left[\left(\frac{B^{(1)}_{\kappa_1,1}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B^{(1)}_{\kappa_2,1}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right)\left(J_{\kappa_1,m}^{(1)}(\lambda_3)+\widehat{S}_{\kappa,1}^{(1)}(\lambda_s)\right)+\frac{B^{(1)}_{\kappa_1,1}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\frac{B^{(1)}_{\kappa_2,1}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right]\\[0.5ex]
    &\qquad +\frac{B_{\kappa_1,-1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\left(\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +\frac{B_{\kappa_2,-1}^{(1)}(x_2,\lambda_1)}{f_j(x_2,\mu_F)}\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +J_{\kappa_3,-1}^{(1)}(\lambda_3)\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +\widehat{S}^{(1)}_{\kappa,-1}\left(\lambda_s\right)\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)\right)\,,
\end{aligned}
\end{equation}
We validate $\mathcal{C}_1^{(2)}/8 = h_{22}$, which follows from
Eq.~\eqref{eq:coeff-convent} with $m=2$, $n=2$. Since $h_{10}$ and
$h_{12}$ were validated at NLO, the cross-term
$\mathcal{C}_1^{(1)}\cdot\mathcal{C}_{-1}^{(1)}$, corresponding to
$h_{10}\cdot h_{12}$ in the \textsc{ARES} cumulant expansion, is
subtracted before the comparison is made
(Fig.~\ref{fig:mathematica-C21-sub}). The output \texttt{0} in
Fig.~\ref{fig:mathematica-h22compare} confirms exact agreement.

\begin{lstlisting}[style=mma, caption={Definition of $\mathcal{C}_1^{(2)}$

four functions at a given order and power of the logarithm.},
label={lst:C21}]
tot[m_, n_] := J[m, n] + B1[m, n] + B2[m, n] + S[m, n];
 
C21 = (tot[2, 1] + H1SCET tot[1, 1]
    + 2 (J[1,0] B1[1,0] + J[1,0] B2[1,0] + B1[1,0] B2[1,0]
         + J[1,0] S[1,0] + B1[1,0] S[1,0] + B2[1,0] S[1,0])
    - \[Pi]^2/3 (B1[1,1] B2[1,1] + B1[1,1] S[1,1] + B2[1,1] S[1,1]
         + B1[1,1] J[1,1] + B2[1,1] J[1,1] + S[1,1] J[1,1])
    + B1[1,1] J[1,-1] + B2[1,1] J[1,-1] + B1[1,1] B2[1,-1]
    + J[1,1] S[1,-1] + B1[1,1] S[1,-1] + B2[1,1] S[1,-1]
    + B1[1,-1] J[1,1] + B2[1,-1] J[1,1] + B1[1,-1] B2[1,1]
    + J[1,-1] S[1,1] + B1[1,-1] S[1,1] + B2[1,-1] S[1,1]);
\end{lstlisting}
 
\begin{lstlisting}[style=mma, caption={Subtraction of the
$\mathcal{C}_1^{(1)}\cdot\mathcal{C}_{-1}^{(1)}$ cross-term from

C11    = tot[1, 1];
C1m1   = tot[1, -1] + H1SCET;

\end{lstlisting}
 
\begin{lstlisting}[style=mma, caption={Definition of $h_{11}$ and $h_{22}$
following Eqs.~\eqref{eq:h11} and~\eqref{eq:h22}.}, label={lst:h22-def}]
h11 = ((g[1,0] + g[2,0] + g[3,0])
    - 2 Ck[1] Log[(Q[1,2]^2 Q[1,3]^2)/(Q[2,3]^2 M Q[1])]
    - 2 Ck[2] Log[(Q[1,2]^2 Q[2,3]^2)/(Q[1,3]^2 M Q[2])]
    - 2 Ck[3] Log[(Q[2,3]^2 Q[1,3]^2)/(Q[1,2]^2 M Q[3])]

    + 2 \[Pi] \[Beta]0 (
        -2 (Ck[1] + Ck[2] - Ck[3]) Log[Q[1,2]^2/Qresum^2]
        -2 (Ck[1] + Ck[3] - Ck[2]) Log[Q[1,3]^2/Qresum^2]
        -2 (Ck[2] + Ck[3] - Ck[1]) Log[Q[2,3]^2/Qresum^2]

        - 3 Ck[2] Log[Qresum^2/(M Q[2])]
        - 3 Ck[3] Log[Qresum^2/(M Q[3])]
        - 1/2 P[1,0] - 1/2 P[2,0])
    + 4 \[Pi] Ctot \[Beta]0 Log[Qresum/\[Mu]R]);
\end{lstlisting}
 
\begin{lstlisting}[style=mma, caption={Comparison of $h_{22}$ and

\section{Validation of \texorpdfstring{$h_{21}$}{h21}}
\label{ap:mathematica-h21}

We now validate $\mathcal{C}^{(2)}_0$, given by
\begin{equation}
\label{eq:C20}
\begin{aligned}
&\mathcal{C}^{(2)}_0=\frac{B_{\kappa_1,0}^{(2)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,0}^{(2)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+
    J_{\kappa_3,0}^{(2)}(\lambda_3)
    +\widehat{S}^{(2)}_{\kappa,0}\left(\lambda_s\right)\\[0.5ex]
    &\qquad + H^{(1)}_{\rm SCET}\left(\frac{B_{\kappa_1,0}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,0}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+
    J_{\kappa_3,0}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,0}\!\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad -\frac{\pi^2}{6}\left[\frac{B_{\kappa_1,0}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\left(\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\right.\\[0.5ex]
    &\qquad+\frac{B_{\kappa_2,0}^{(1)}(x_2,\lambda_1)}{f_j(x_2,\mu_F)}\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +J_{\kappa_3,0}^{(1)}(\lambda_3)\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+\widehat{S}^{(1)}_{\kappa,1}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad \left.+\widehat{S}^{(1)}_{\kappa,0}\left(\lambda_s\right)\left(\frac{B_{\kappa_1,1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,1}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,1}^{(1)}(\lambda_3)\right)\right]\\[0.5ex]
    &\qquad +2\zeta_3\left[\left(\frac{B^{(1)}_{\kappa_1,1}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B^{(1)}_{\kappa_2,1}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right)\left(J_{\kappa_1,1}^{(1)}(\lambda_3)+\widehat{S}_{\kappa,1}^{(1)}(\lambda_s)\right)+\frac{B^{(1)}_{\kappa_1,1}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\frac{B^{(1)}_{\kappa_2,1}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}\right]\\[0.5ex]
    &\qquad +\frac{B_{\kappa_1,-1}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}\left(\frac{B_{\kappa_2,0}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,0}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,0}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +\frac{B_{\kappa_2,-1}^{(1)}(x_2,\lambda_1)}{f_j(x_2,\mu_F)}\left(\frac{B_{\kappa_1,0}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+J_{\kappa_3,0}^{(1)}(\lambda_3)
    +\widehat{S}^{(1)}_{\kappa,0}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +J_{\kappa_3,-1}^{(1)}(\lambda_3)\left(\frac{B_{\kappa_1,0}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,0}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+\widehat{S}^{(1)}_{\kappa,0}\left(\lambda_s\right)\right)\\[0.5ex]
    &\qquad +\widehat{S}^{(1)}_{\kappa,-1}\left(\lambda_s\right)\left(\frac{B_{\kappa_1,0}^{(1)}(x_1,\lambda_1)}{f_i(x_1,\mu_F)}+\frac{B_{\kappa_2,0}^{(1)}(x_2,\lambda_2)}{f_j(x_2,\mu_F)}+J_{\kappa_3,0}^{(1)}(\lambda_3)\right)\,.
\end{aligned}
\end{equation}

We validate $\mathcal{C}_0^{(2)}/(-4) = h_{21}$, which follows from
Eq.~\eqref{eq:coeff-convent} with $m=2$, $n=1$. As above, the cross-term
$\mathcal{C}_0^{(1)}\cdot\mathcal{C}_{-1}^{(1)}$, corresponding to
$h_{10}\cdot h_{11}$, is subtracted before the comparison is made
(Listing~\ref{lst:C20-sub}). In carrying out the comparison we substitute the
explicit expression for $\delta\mathcal{F}_{\rm wa}$, together with the
relation
\begin{equation}
\label{eq:Chc-map}
\begin{aligned}
 \sum_{\ell}C_{\mathrm{hc},\ell}^{(1)}&+\delta\mathcal{F}_{\rm rec}
 = \sum_{k}\int\frac{dz}{z} \frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}2I_{ik}^{(1)}
 +\sum_{k}\int\frac{dz}{z}\frac{f_k(x_2/z,\mu_F)}{f_j(x_2,\mu_F)}2I_{jk}^{(1)}\\
	&\qquad+\sum_{k}\int\frac{dz}{z}\left(P_{ik}(z)\ln\frac{Q_1\,M}{Q^2}
    +\gamma_{1}^{(0)}\ln\frac{Q^2}{Q_1\,M}\right)
    \frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}\\
    &\qquad+\sum_{k}\int\frac{dz}{z}\left(P_{jk}(z)\ln\frac{Q_2\,M}{Q^2}
    +\gamma_{2}^{(0)}\ln\frac{Q^2}{Q_2\,M}\right)
    \frac{f_k(x_2/z,\mu_F)}{f_j(x_2,\mu_F)}\\
& \qquad+ J_{\kappa_3,-1}^{(1)}+\gamma_{3}^{(0)}\ln\frac{Q^2}{Q_3\,M}
 + \sum_{\ell}C_{\ell}\frac{\pi^2}{6}\,,
 \end{aligned}
\end{equation}
for the hard-collinear coefficients. The comparison then leaves a residual
proportional to $\gamma^{(1)}_\ell$, which does not vanish symbolically since
the SCET and \textsc{ARES} conventions for $\gamma^{(1)}$ differ.
Substituting explicit values for the $q\bar{q}\to Zg$ channel, the residual
vanishes exactly (Listing~\ref{lst:h21-compare}), where the other channels follow
in the same way.
