\chapter{Coefficients and subtraction ingredients for one-jettiness}
\label{app:ch5}

In this appendix we present the formulae entering the validation
shown in Sec.~\ref{subsec:N-jet-SCET-val}, namely the explicit form of the 
\textsc{ARES} resummation coefficients~\cite{Banfi:2004yd} for one-jettiness
together with the ingredients required to build the
associated \textsc{SCET} subtraction terms of Ref.~\cite{Gaunt:2015pea}. 
Here we present the components used to build the subtraction ingredients
of Ref.~\cite{Gaunt:2015pea}.

\section{\textsc{ARES} coefficients}
\label{ap:ARES-coeff}

The coefficients of $\mathcal{O}(\alpha_s)$ are:
\begin{subequations}
\begin{align}
\label{eq:h12}
&h_{12} =-\sum_{\ell}C_{\ell}\,,\\[1ex]
\label{eq:h11}
&h_{11} = \sum_{\ell}\left(\gamma_{\ell}^{(0)}-2\sum_{\ell}\sum_{\bar\ell\neq\ell} C_{\ell\bar\ell}\ln  d^{(\ell\bar{\ell})}_\ell\right)\notag\\[1ex]
&\qquad-\sum_{k}\int\frac{dz}{z}P_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\sum_{k}\int\frac{dz}{z}P_{jk}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}\,,\\[1.5ex]
\label{eq:h10}
&h_{10}= H^{(1)}-\sum_{\ell}\sum_{\bar\ell\neq\ell} C_{\ell\bar\ell}\ln^2 d_\ell^{(\ell\bar\ell)}+\frac{\alpha_s(Q)}{\alpha_s(\tau_1 Q)}\frac{\delta\mathcal{F}_{\rm wa}}{\FNLL}+ \sum_{\ell}C_{\mathrm{hc},\ell}^{(1)}+\frac{\alpha_s(Q)}{\alpha_s\left(Q\sqrt{\tau_1}\right)}\frac{\delta\mathcal{F}_{\rm rec}}{\FNLL}\notag\\[0.5ex]
&\qquad +\sum_{k}\int\frac{dz}{z}\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)} P_{ik}(z)\ln\frac{Q^2}{\mu_F^2}
+\sum_{k}\int\frac{dz}{z}\frac{f_k(x_1/z,\mu_F)}{f_j(x_2,\mu_F)} P_{jk}(z)\ln\frac{Q^2}{\mu_F^2}\,.
\end{align}
\end{subequations}

The coefficients of $\mathcal{O}(\alpha_s^2)$ are:
\begin{subequations}
\begin{align}
\label{eq:h24}
&h_{24}=\frac{1}{2}\left(\sum_{\ell}C_{\ell}\right)^2\,,\\[1.5ex]
\label{eq:h23}
&h_{23}=-2\pi\beta_{0} \sum_{\ell}C_{\ell }-h_{11} \sum_{\ell}C_{\ell }\,,\\[1.5ex]
\label{eq:h22}
&h_{22}=h_{10}\cdot h_{12}+\frac{1}{2}h_{11}^2+4\pi\ln\frac{\mu_R}{Q}h_{12}-\frac{\pi^2}{3}\left(\sum_{\ell}C_{\ell}\right)^2 -K^{(1)}\sum_{\ell}C_{\ell} \notag\\[0.5ex]
&\qquad+2\pi\beta_0\left[-4\sum_{\bar\ell\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar\ell}^2}{Q^2}
+\sum_{\ell}\left(\frac{\gamma_\ell^{(0)}}{2}-3C_\ell \ln\frac{Q^2}{M Q_{\ell}}\right)\right. \notag \\[0.5ex]
&\qquad\qquad\qquad\left. -\frac{1}{2}\sum_{k}\int\frac{dz}{z}P_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\frac{1}{2}\sum_{k}\int\frac{dz}{z}P_{jk}(z)\frac{f_k(x_2/z,\mu_F)}{f_j(x_2,\mu_F)}\right]\,,\\[1.5ex]
\label{eq:h21}
&h_{21}=h_{11}\cdot h_{10}+4\pi\beta_0\ln\frac{\mu_R}{Q}h_{11}-\sum_{\ell}\gamma_{\ell}^{(1)}+ 2\left(\sum_{\ell}C_{\ell}\right)^2\psi^{(2)}(1)+2\pi\beta_0\frac{\pi^2}{6}\sum_{\ell}C_{\ell} \notag\\[0.5ex]
&\qquad+\sum_{\ell}\sum_{\ell\neq \bar{\ell}} C_{\ell\bar{\ell}}\left(-2K^{(1)} \ln d_\ell^{(\ell\bar{\ell})}
  - 6\pi\beta_0 \ln^2 d_\ell^{(\ell\bar{\ell})} - \frac{\pi^3\beta_0}{3}+2 \pi\beta_0 \ln d_\ell^{(\ell\bar{\ell})}
 \ln\frac{Q_{\ell\bar{\ell}}^2}{Q^2}\right) \notag \\[0.5ex]
 &\qquad+2\pi\beta_0\left(\frac{\alpha_s(Q)}{\alpha_s(\tau_1 Q)}\frac{\delta\mathcal{F}_{\rm wa}}{\FNLL}+ \sum_{\ell}C_{\mathrm{hc},\ell}^{(1)}+\frac{\alpha_s(Q)}{\alpha_s\left(Q\sqrt{\tau_1}\right)}\frac{\delta\mathcal{F}_{\rm rec}}{\FNLL}\right)\notag\\[0.5ex]
&\qquad-\frac{\pi^2}{3} \sum_{\ell}C_{\ell}\left[\sum_{\ell}\left(B_{\ell}+C_{\ell}\ln\frac{Q^2 M}{Q_{\ell}}+ 8C_{\ell\bar\ell}\ln \frac{Q_{\ell\bar\ell}}{Q}\right) \right]\notag\\[0.5ex]
&\qquad-\sum_{k}\int\frac{dz}{z}P^{(1)}_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\sum_{k}\int\frac{dz}{z}P^{(1)}_{jk}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}\,.
\end{align}
\end{subequations}

The matching functions, $I_{ak}^{(1)}$ and $I_{bk}^{(1)}$, are expressed as:
\begin{subequations}
\begin{align}
  I^{(1)}_{qq}(z) &=C_F\left[ \left(\frac{\ln(1-z)}{1-z}\right)_+(1+z^2)
  -\frac{\pi^2}{6}\delta(1-z)+\Theta(1-z)\left(1-z-\frac{1+z^2}{1-z}\ln z\right)\right]\,,\\[1ex]
  I^{(1)}_{qg}(z) &= T_R\left[\left(z^2+(1-z)^2\right)\ln\frac{1-z}{z}
  +2z(1-z)\right]\,,\\[1ex]
  I^{(1)}_{gq}(z) &= C_F\left[\frac{1+(1-z)^2}{z}\ln\frac{1-z}{z}+z\right]\,,\\[1ex]
  I^{(1)}_{gg}(z) &= C_A\left[\left(\frac{\ln(1-z)}{1-z}\right)_+\frac{2(1-z+z^2)^2}{z}-\frac{\pi^2}{6}\delta(1-z)-\frac{2\ln z}{(1-z)_+}\frac{(1-z+z^2)^2}{z}\right]\notag\\[0.5ex]
  &\qquad\qquad+\frac{\beta_0}{2}\delta(1-z)\,.
\end{align}
\end{subequations}
