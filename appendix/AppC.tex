\chapter{One-jettiness resummation coefficients}
\label{ap:ARES-coeff}

In this appendix we collect the ingredients entering the validation of
the one-jettiness resummation presented in Sec.~\ref{subsec:N-jet-SCET-val}.
We first give the expansion coefficients of the \textsc{ARES} resummation
formula, followed by the corresponding $N$-jettiness SCET ingredients
used to construct the subtraction coefficients extracted from Ref.~\cite{Gaunt:2015pea}.

\section{ARES expansion coefficients}
The \textsc{ARES} expansion coefficients entering Eq.~\eqref{eq:sig-coeff}
for one-jettiness are given below.

The LL coefficients are:
\begin{subequations}
\begin{align}
\label{eq:h12}
&h_{12} =-\sum_{\ell}C_{\ell}\,,\\[1ex]
\label{eq:h24}
&h_{24}=\frac{1}{2}\left(\sum_{\ell}C_{\ell}\right)^2\,.
 \end{align}
\end{subequations}

The NLL coefficients are: 
\begin{subequations}
\begin{align}
\label{eq:h11}
&h_{11} = \sum_{\ell}\left(\gamma_{\ell}^{(0)}-2\sum_{\ell}\sum_{\bar\ell\neq\ell} C_{\ell\bar\ell}\ln  d^{(\ell\bar{\ell})}_\ell\right)\notag\\[1ex]
&\qquad-\sum_{k}\int\frac{dz}{z}P_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\sum_{k}\int\frac{dz}{z}P_{jk}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}\,,\\[1.5ex]
\label{eq:h23}
&h_{23}=-2\pi\beta_{0} \sum_{\ell}C_{\ell }-h_{11} \sum_{\ell}C_{\ell }\,.
 \end{align}
\end{subequations}

The NNLL coefficients are:
\begin{subequations}
\begin{align}
&h_{10}= H_{ij}^{(1)}-\sum_{\ell}\sum_{\bar\ell\neq\ell} C_{\ell\bar\ell}\ln^2 d_\ell^{(\ell\bar\ell)}+\frac{\alpha_s(Q)}{\alpha_s(\tau_1 Q)}\frac{\delta\mathcal{F}_{\rm wa}}{\FNLL}+ \sum_{\ell}C_{\mathrm{hc},\ell}^{(1)}+\frac{\alpha_s(Q)}{\alpha_s\left(Q\sqrt{\tau_1}\right)}\frac{\delta\mathcal{F}_{\rm rec}}{\FNLL}\notag\\[0.5ex]
&\qquad +\sum_{k}\int\frac{dz}{z}\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)} P_{ik}(z)\ln\frac{Q^2}{\mu_F^2}
+\sum_{k}\int\frac{dz}{z}\frac{f_k(x_1/z,\mu_F)}{f_j(x_2,\mu_F)} P_{jk}(z)\ln\frac{Q^2}{\mu_F^2}\,,\\[1.5ex]
\label{eq:h22}
&h_{22}=h_{10}\cdot h_{12}+\frac{1}{2}h_{11}^2+4\pi\ln\frac{\mu_R}{Q}h_{12}-\frac{\pi^2}{3}\left(\sum_{\ell}C_{\ell}\right)^2 -K^{(1)}\sum_{\ell}C_{\ell} \notag\\[0.5ex]
&\qquad+2\pi\beta_0\left[-4\sum_{\bar\ell\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar\ell}^2}{Q^2}
+\sum_{\ell}\left(\frac{\gamma_\ell^{(0)}}{2}-3C_\ell \ln\frac{Q^2}{M Q_{\ell}}\right)\right. \notag \\[0.5ex]
&\qquad\qquad\qquad\left. -\frac{1}{2}\sum_{k}\int\frac{dz}{z}P_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\frac{1}{2}\sum_{k}\int\frac{dz}{z}P_{jk}(z)\frac{f_k(x_2/z,\mu_F)}{f_j(x_2,\mu_F)}\right]\,,\\[1.5ex]
\label{eq:h21}
&h_{21}=h_{11}\cdot h_{10}+4\pi\beta_0\ln\frac{\mu_R}{Q}h_{11}-\sum_{\ell}\gamma_{\ell}^{(1)}+ 2\left(\sum_{\ell}C_{\ell}\right)^2\psi^{(2)}(1)+2\pi\beta_0\frac{\pi^2}{6}\sum_{\ell}C_{\ell} \notag\\[0.5ex]
&\qquad+\sum_{\ell}\sum_{\ell\neq \bar{\ell}} C_{\ell\bar{\ell}}\left(-2K^{(1)} \ln d_\ell^{(\ell\bar{\ell})}
  - 6\pi\beta_0 \ln^2 d_\ell^{(\ell\bar{\ell})} - \frac{\pi^3\beta_0}{3}+2 \pi\beta_0 \ln d_\ell^{(\ell\bar{\ell})}
 \ln\frac{Q_{\ell\bar{\ell}}^2}{Q^2}\right) \notag \\[0.5ex]
 &\qquad+2\pi\beta_0\left(\frac{\alpha_s(Q)}{\alpha_s(\tau_1 Q)}\frac{\delta\mathcal{F}_{\rm wa}}{\FNLL}+ \sum_{\ell}C_{\mathrm{hc},\ell}^{(1)}+\frac{\alpha_s(Q)}{\alpha_s\left(Q\sqrt{\tau_1}\right)}\frac{\delta\mathcal{F}_{\rm rec}}{\FNLL}\right)\notag\\[0.5ex]
&\qquad-\frac{\pi^2}{3} \sum_{\ell}C_{\ell}\left[\sum_{\ell}\left(B_{\ell}+C_{\ell}\ln\frac{Q^2 M}{Q_{\ell}}+ 8C_{\ell\bar\ell}\ln \frac{Q_{\ell\bar\ell}}{Q}\right) \right]\notag\\[0.5ex]
&\qquad-\sum_{k}\int\frac{dz}{z}P^{(1)}_{ik}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}-\sum_{k}\int\frac{dz}{z}P^{(1)}_{jk}(z)\frac{f_k(x_1/z,\mu_F)}{f_i(x_1,\mu_F)}\,.
\end{align}
\end{subequations}


\section{SCET subtraction ingredients}
\label{ap:subtract-ingred}

The set of partons is labelled
$\kappa\equiv\{\kappa_1,\kappa_2,\kappa_3\}$, where $\kappa_1$ and
$\kappa_2$ denote the incoming partons on legs $\ell=1,2$, with flavours
$i$ and $j$, respectively, while $\kappa_3$ denotes the outgoing parton
on leg $\ell=3$.  \todo{Need to reformat.}

The ingredients for the LL coefficients are:
\begin{subequations}
\label{eq:C11-ingred}
\begin{align}
&B_{\kappa_{1},1}^{(1)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z} 4C_1\,\delta_{ik}\delta(1-z) f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},1}^{(1)}(x_2,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z} 4C_2\,\delta_{jk}\delta(1-z)  f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
&J_{\kappa_3,1}^{(1)}(\lambda_3) = 4C_3\,, \\[1ex]
&\widehat{S}_{\kappa,1}^{(1)} \left(\{\hat{p}_{\ell}\},\lambda_s\right)= -8\sum_{\ell}C_{\ell}\,.
\end{align}
\end{subequations}

\begin{subequations}
\label{eq:C23-ingred}
\begin{align}
&B_{\kappa_{1},3}^{(2)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z}
8C_1^2\,\delta_{ik}\delta(1-z) f_k\left(\frac{x_1}{z},\mu_F\right) \\[1ex]
&B_{\kappa_{2},3}^{(2)}(x_2,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z}
8C_2^2\,\delta_{jk}\delta(1-z) f_k\left(\frac{x_2}{z},\mu_F\right) \\[1ex]
&J_{\kappa_3,3}^{(2)}(\lambda_3) = 8C_3^2\,, \\[1ex]
&\widehat{S}_{\kappa,3}^{(2)}\left(\{\hat{p}_{\ell}\},\lambda_s\right) = 32\left(\sum_{\ell}C_{\ell}\right)^2\,.
\end{align}
\end{subequations}

The ingredients first entering the NLL coefficients are:
\begin{subequations}
\label{eq:C10-ingred}
\begin{align}
&B_{\kappa_{1},0}^{(1)}(x_1,\mu_R,\mu_F,\lambda_1)=\sum_k \int\frac{dz}{z} \left(2P_{ik}(z)-2\gamma_1^{(0)}\,\delta_{ik}\delta(1-z)\right)
f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},0}^{(1)}(x_2,\mu_R,\mu_F,\lambda_2)  =\sum_k \int\frac{dz}{z} \left(2P_{jk}(z)-2\gamma_2^{(0)}\,\delta_{jk}\delta(1-z)\right)
f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
&J_{\kappa_3,0}^{(1)}(\lambda_3) = -2\gamma_3^{(0)}\,, \\[1ex]
&\widehat{S}_{\kappa,0}^{(1)}\left(\{\hat{p}_{\ell}\},\lambda_s\right)=4\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}
\end{align}
\end{subequations}
where $P_{ik}(z), P_{jk}(z)$ are the Altarelli-Parisi splitting functions found in Eq.~\eqref{eq:full-AP}
and $\gamma^{(0)}_{\ell}$ are given in Eq.~\eqref{eq:gammal-flav}.

\begin{subequations}
\label{eq:C22-ingred}
\begin{align}
&B_{\kappa_{1},2}^{(2)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z}\left\{4C_1\left[
    3P_{ik}(z) -\left(3\gamma_1^{(0)}+2\pi\beta_0\right)\delta_{ik}\delta(1-z)
   \right]\right\}f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},2}^{(2)}(x_1,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z}\left\{4C_2\left[
    3P_{jk}(z) -\left(3\gamma_2^{(0)}+2\pi\beta_0\right)\delta_{jk}\delta(1-z)
   \right]\right\}f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
 &J_{\kappa_3,2}^{(2)}(\lambda_3) = -2C_3\left(6\gamma_3^{(0)}+4\pi\beta_0\right)\,, \\[1ex]
&\widehat{S}_{\kappa,2}^{(2)}\left(\{\hat{p}_{\ell}\},\lambda_s\right)  =
    4\sum_{\ell}C_{\ell}\left(8\pi\beta_0-12\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}\right)\,.
\end{align}
\end{subequations}

The ingredients first entering the NNLL coefficients are:
\begin{subequations}
\label{eq:Cd1-ingred}
\begin{align}
&B_{\kappa_{1},-1}^{(1)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z} \left( 2I_{ik}^{(1)}(z)+2\ln\frac{\mu_R^2}{\mu_F^2}\,P_{ik}(z)\right)f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},-1}^{(1)}(x_2,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z} \left( 2I_{jk}^{(1)}(z)+2\ln\frac{\mu_R^2}{\mu_F^2}\,P_{jk}(z)\right)f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
&J_{\kappa_3,-1}^{(1)} = \begin{cases}
    C_F(7-\pi^2)\,, & \text{if $\ell=3$ is a quark}\,, \\[4pt]
    C_A\!\left(\dfrac{4}{3}-\pi^2\right)+\dfrac{20}{3}\pi\beta_0\,, &\text{if $\ell=3$ is a gluon}\,,
\end{cases} \\[1ex]
&\widehat{S}_{\kappa,-1}^{(1)}\left(\{\hat{p}_{\ell}\},\lambda_s\right)
  = -\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\left(\ln^2\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}-\frac{\pi^2}{6}+4\sum_{\hat{\ell}\neq\ell,\bar{\ell}}I_{\ell\bar{\ell},\hat{\ell}}\right)\,.
\end{align}
\end{subequations}
where the matching functions, $I_{ak}^{(1)}(z)$ and $I_{bk}^{(1)}(z)$, 
are defined in Refs.~\cite{Gaunt:2014xga,Gaunt:2014cfa}.

\begin{subequations}
\label{eq:C21-ingred}
\begin{align}
&B_{\kappa_{1},1}^{(2)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z}
   \left\{8 \left[C_1 K^{(1)}-\frac{\pi^2}{3}C_1^2
    +\gamma_1^{(0)}\left(\frac{\gamma_1^{(0)}}{2}+\pi\beta_0\right)\right]
    \delta_{ik}\delta(1-z) \right.\notag\\[0.5ex]
  &\qquad\qquad\qquad\qquad\qquad+8C_1\left(I_{ik}^{(1)}(z)+\ln\frac{\mu_R^2}{\mu_F^2}\,P_{ik}(z)\right)
    +4\sum_m P_{im}(z)\otimes_z P_{mk}(z)\notag\\
    &\qquad\qquad\qquad\qquad\qquad\left.-2\left(4\gamma_1^{(0)}+4\pi\beta_0\right)P_{ik}(z)\right\}f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},1}^{(2)}(x_2,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z}
   \left\{8 \left[C_2 K^{(1)}-\frac{\pi^2}{3}C_2^2
    +\gamma_2^{(0)}\left(\frac{\gamma_2^{(0)}}{2}+\pi\beta_0\right)\right]
    \delta_{jk}\delta(1-z) \right.\notag\\[0.5ex]
  &\qquad\qquad\qquad\qquad\qquad+8C_2\left(I_{jk}^{(1)}(z)+\ln\frac{\mu_R^2}{\mu_F^2}\,P_{jk}(z)\right)
    +4\sum_m P_{jm}(z)\otimes_z P_{mk}(z)\notag\\
    &\qquad\qquad\qquad\qquad\qquad\left.-2\left(4\gamma_2^{(0)}+4\pi\beta_0\right)P_{jk}(z)\right\}f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
    &J_{\kappa_3,1}^{(2)}(\lambda_3) = 8C_3 K^{(1)}-\frac{8\pi^2}{3}C_3^2
    +2\gamma_3^{(0)}\!\left(2\gamma_3^{(0)}+4\pi\beta_0\right)
    +4C_3 J_{\kappa_3,-1}^{(1)}\,, \\[1ex]
&\widehat{S}_{\kappa,1}^{(2)}\left(\{\hat{p}_{\ell}\},\lambda_s\right) =16\left(\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}\right)^2
-32\pi\beta_0\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}\notag\\
&\qquad\qquad\qquad -\frac{32\pi^2}{3}\left(\sum_{\ell}C_{\ell}\right)^2-\left(8\widehat{S}_{\kappa,-1}^{(1)}+16K^{(1)}\right)\sum_{\ell}C_{\ell}\,.
\end{align}
\end{subequations}

\begin{subequations}
\label{eq:C20-ingred}
\begin{align}
&B_{\kappa_{1},0}^{(2)}(x_1,\mu_R,\mu_F,\lambda_1) = \sum_{k}\int\frac{dz}{z}
   \left\{ \left(16C_1^2\zeta_3+\frac{4\pi^2}{3}C_1\gamma_1^{(0)}
    -\frac{\gamma_{1,s}^{(1)}}{2}\right)\delta_{ik}\delta(1-z)
    -\frac{4\pi^2}{3}C_1 P_{ik}(z) \right.\notag\\[0.5ex]
  &\qquad-\left(4\gamma_1^{(0)}+8\pi\beta_0\right)I_{ik}^{(1)}(z)
    +4\sum_m I_{im}^{(1)}(z)\otimes_z P_{mk}(z)+4P_{ik}^{(1)}(z) \notag\\[0.5ex]
  &\qquad\left.+\ln\frac{\mu_R^2}{\mu_F^2}\left[-4\gamma_1^{(0)}P_{ik}(z)
    +4\sum_m P_{im}(z)\otimes_z P_{mk}(z)+4P_{ik}^{(1)}(z)\right]\right\}f_k\left(\frac{x_1}{z},\mu_F\right)\,, \\[1ex]
&B_{\kappa_{2},0}^{(2)}(x_2,\mu_R,\mu_F,\lambda_2) = \sum_{k}\int\frac{dz}{z}
   \left\{ \left(16C_2^2\zeta_3+\frac{4\pi^2}{3}C_2\gamma_2^{(0)}
    -\frac{\gamma_{2,s}^{(1)}}{2}\right)\delta_{jk}\,\delta(1-z)
    -\frac{4\pi^2}{3}C_2 P_{jk}(z) \right.\notag\\[0.5ex]
  &\qquad-\left(4\gamma_2^{(0)}+8\pi\beta_0\right)I_{jk}^{(1)}(z)
    +4\sum_m I_{jm}^{(1)}(z)\otimes_z P_{mk}(z)+4P_{jk}^{(1)}(z) \notag\\[0.5ex]
  &\qquad\left.+\ln\frac{\mu_R^2}{\mu_F^2}\left[-4\gamma_2^{(0)}P_{jk}(z)
    +4\sum_m P_{jm}(z)\otimes_z P_{mk}(z)+4P_{jk}^{(1)}(z)\right]\right\}f_k\left(\frac{x_2}{z},\mu_F\right)\,, \\[1ex]
    &J_{\kappa_3,0}^{(2)}(\lambda_3) = 16C_3^2\zeta_3+\frac{4\pi^2}{3}C_3\gamma_3^{(0)}
    -\left(4\gamma_3^{(0)}+4\pi\beta_0\right)J_{\kappa_3,-1}^{(1)}
    -\frac{\gamma_{3,s}^{(1)}}{2}\,, \\[1ex]
&\widehat{S}_{\kappa,0}^{(2)} =
    64\zeta_3\left(\sum_{\ell}C_{\ell}\right)^2-\gamma_S^{(1)}\sum_{\ell}C_{\ell}
    -8\pi\beta_0\,\widehat{S}_{\kappa,-1}^{(1)} \notag\\[0.5ex]
  &\qquad+4\sum_{\ell}\sum_{\bar{\ell}\neq\ell}C_{\ell\bar\ell}\ln\frac{Q_{\ell\bar{\ell}}^2}{Q_{\ell}\,Q_{\bar{\ell}}}\left(\frac{4\pi^2}{3}\sum_{\ell}C_{\ell}+\widehat{S}_{\kappa,-1}^{(1)}+K^{(1)}\right)\,.
\end{align}
\end{subequations}
The convolution $\otimes_z$ is defined as:
\begin{equation}
P_{ij}(z)\otimes_z P_{jk}(z)=\int_z^1\frac{dw}{w}P_{ij}(w)P_{jk}\left(\frac{z}{w}\right)
\end{equation}
and analogously for $I_{ij}^{(1)}\otimes_z P_{jk}$, where the index $j$ is not summed over. 
Additionally,  the noncusp soft anomalous dimension is
\begin{equation}
\gamma_S^{(1)}=C_A\left(-\frac{64}{9}+28\zeta_3\right)+4\pi\beta_0\left(-\frac{56}{9}+\frac{\pi^2}{3}\right)\,,
\end{equation}
while those for the beam and jet functions $\gamma^{(1)}_{\ell,s}$ are, for a quark,
\begin{equation}
\gamma_{\ell,s}^{(1)}=C_F\left[C_A\left(\frac{146}{9}-80\zeta_3\right)+C_F(3-4\pi^2+48\zeta_3)+4\pi\beta_0\left(\frac{121}{9}+\frac{2\pi^2}{3}\right)\right]\,,
\end{equation}
and for a gluon,
\begin{equation}
\gamma_{\ell,s}^{(1)}=C_A\left[C_A\left(\frac{182}{9}-32\zeta_3\right)+4\pi\beta_0\left(\frac{94}{9}-\frac{2\pi^2}{3}\right)\right]\,.
\end{equation}
The subscript $s$ indicates that these anomalous dimensions follow the
$N$-jettiness SCET convention, which differs from the convention used in
the \textsc{ARES} formalism, where the corresponding quantities are defined
in Eq.~\eqref{eq:gammal1-flav}.
