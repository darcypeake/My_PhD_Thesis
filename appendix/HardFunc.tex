\chapter{One-loop hard function}
\label{ap:H1-extraction}

Following the argument of Sec.~\ref{subsec:NNLL-coeff}, we extract the
hard coefficient $H^{(1)}_{ij}$ of Eq.~\eqref{eq:HQ-expanded} and the
one-loop hard function $H^{(1)}_{\rm SCET}$ of Eq.~\eqref{eq:CH-SCET}
for the $q\bar q\to Hg$ channel, using the known one-loop virtual
corrections to $hh\to H+\mathrm{jet}$~\cite{Ravindran:2002dc}. The relation
between the two hard functions, like the infrared poles themselves, is fixed
by the colour charges and directions of the external partons
(cf.\ Eq.~\eqref{eq:HQ}). Since $H+\mathrm{jet}$ and $Z+\mathrm{jet}$
production share the same colour configuration, the Higgs amplitude serves to
determine both. The remaining partonic channels follow analogously.

Adopting the convention given in Sec.~\ref{sec:H1},
the one-loop virtual corrections for $q\bar q\to Hg$ are
\begin{equation}
\label{eq:V-oneloop}
\begin{aligned}
&\mathcal{V}(Q;\varepsilon)=1+\frac{\alpha_s(\mu_R)}{2\pi}\left[
-\frac{C_A+2C_F}{\varepsilon^2}+\frac{1}{\varepsilon}\left(-\frac{11}{6}C_A-3C_F
+\frac{n_f}{3}+(C_A-2C_F)\ln\frac{\mu_R^2}{Q_{12}^2}\right.\right. \\[0.5ex]
&\qquad\qquad\qquad \left. 
-C_A\ln\frac{\mu_R^2}{Q_{13}^2}-C_A\ln\frac{\mu_R^2}{Q_{23}^2}\right)
-n_f\left(\frac{5}{9}+\frac{1}{3}\ln\frac{\mu_R^2}{Q_{12}^2}\right)+\frac{\pi^2}{12}\left(
18C_F-7C_A\right)\\
& \qquad\qquad\qquad+C_A\left(
\frac{38}{9}+\operatorname{Li}_2\left(\frac{Q_{12}^2-M^2}{Q_{12}^2}\right)
-\frac{11}{6}\ln\frac{Q_{12}^2}{\mu_R^2}+\frac{1}{4}\ln^2\frac{\mu_R^2}{Q_{12}^2}
-\frac{1}{4}\ln^2\frac{\mu_R^2}{Q_{13}^2}\right.\\[0.5ex]
&\qquad\qquad\qquad\left.-\frac{1}{4}\ln^2\frac{\mu_R^2}{Q_{23}^2}
+\frac{1}{2}\ln\frac{Q_{12}^2}{Q_{13}^2} \ln\frac{Q_{12}^2}{Q_{23}^2}\right)
+C_F\left(-4+\operatorname{Li}_2\left(\frac{-Q_{13}^2}{M^2}\right)
+\operatorname{Li}_2\left(\frac{-Q_{23}^2}{M^2}\right) \right.\\[0.5ex]
&\qquad\qquad\qquad-\frac{1}{2}\ln^2\frac{Q_{13}^2+M^2}{Q_{13}^2}
-\frac{1}{2}\ln^2\frac{Q_{23}^2+M^2}{Q_{23}^2}
+\frac{1}{2}\ln^2\frac{M^2+Q_{13}^2}{M^2}
+\frac{1}{2}\ln^2\frac{M^2+Q_{23}^2}{M^2}\\[0.5ex]
&\qquad\qquad\qquad\left.\left.
-\frac{1}{2}\ln^2\frac{Q_{12}^2}{\mu_R^2}
+\frac{3}{2}\ln\frac{Q_{12}^2}{\mu_R^2}
+\frac{1}{2}\ln^2\frac{Q_{13}^2}{Q_{23}^2}
\right)\right]\,.
\end{aligned}
\end{equation}

The SCET hard function is obtained from the finite terms of
Eq.~\eqref{eq:V-oneloop}. By the definition in Eq.~\eqref{eq:CH-SCET},
it is twice the one-loop correction,
\begin{equation}
\label{eq:H1-SCET}
\begin{aligned}
& H^{(1)}_{\rm SCET}=\pi^2\left(
3C_F-\frac{7}{6}C_A\right)-n_f\left(\frac{10}{9}+\frac{2}{3}\ln\frac{\mu_R^2}{Q_{12}^2}\right)+C_A\left(\frac{76}{9}
+2\operatorname{Li}_2\left(\frac{Q_{12}^2-M^2}{Q_{12}^2}
\right)\right.\\[0.5ex]
& \qquad\qquad\left.-\frac{11}{3}\ln\frac{Q_{12}^2}{\mu_R^2}
+\frac{1}{2}\ln^2\frac{\mu_R^2}{Q_{12}^2}
-\frac{1}{2}\ln^2\frac{\mu_R^2}{Q_{13}^2}
-\frac{1}{2}\ln^2\frac{\mu_R^2}{Q_{23}^2}+\ln\frac{Q_{12}^2}{Q_{13}^2}
 \ln\frac{Q_{12}^2}{Q_{23}^2}
\right)\\[0.5ex]
&\qquad\qquad+C_F\left(-8
+2\operatorname{Li}_2\left(\frac{-Q_{13}^2}{M^2}\right)
+2\operatorname{Li}_2\left(\frac{-Q_{23}^2}{M^2}\right)
-\ln^2\frac{Q_{13}^2+M^2}{Q_{13}^2}
-\ln^2\frac{Q_{23}^2+M^2}{Q_{23}^2}\right.\\
&\qquad\qquad\left.
+\ln^2\frac{M^2+Q_{13}^2}{M^2}
+\ln^2\frac{M^2+Q_{23}^2}{M^2}
-\ln^2\frac{Q_{12}^2}{\mu_R^2}
+3\ln\frac{Q_{12}^2}{\mu_R^2}
+\ln^2\frac{Q_{13}^2}{Q_{23}^2}
\right)\,.
\end{aligned}
\end{equation}

While $H^{(1)}_{\rm SCET}$ retains all finite terms of the virtual correction,
the \textsc{ARES} coefficient does not. Instead, starting from the finite
terms of Eq.~\eqref{eq:V-oneloop}, we subtract the finite remainder of the
pole structure in Eq.~\eqref{eq:HQ-expanded}. Using Eq.~\eqref{eq:H1-final},
the \textsc{ARES} hard coefficient for the $q\bar{q}$ channel is
\begin{equation}
H_{q\bar{q}}^{(1)}=\mathcal{H}_1
+3\left(C_F-\frac{C_A}{2}\right)\ln\frac{Q_{12}^2}{Q^2}
+\left(\frac{3C_A}{4}+\pi\beta_0\right)\ln\frac{Q_{13}^2}{Q^2}
+\left(\frac{3C_A}{4}+\pi\beta_0\right)\ln\frac{Q_{23}^2}{Q^2}\,,
\end{equation}
where
\begin{equation}
\label{eq:H1-ARES}
\begin{aligned}
&\mathcal{H}_1=\frac{2\pi^2}{3}\left(2C_F-C_A\right)
-n_f\left(\frac{5}{9}+\frac{1}{3}\ln\frac{\mu_R^2}{Q_{12}^2}\right)
+\pi\beta_0\left(\ln\frac{\mu_R^2}{Q_{13}^2}+\ln\frac{\mu_R^2}{Q_{23}^2}\right)\\[0.5ex]
&\qquad+C_A\left(\frac{38}{9}
+\operatorname{Li}_2\left(\frac{Q_{12}^2-M^2}{Q_{12}^2}\right)
+\frac{1}{4}\ln^2\frac{Q_{13}^2Q_{23}^2}{\mu_R^2\,Q_{12}^2}
+\frac{1}{3}\ln\frac{\mu_R^2}{Q_{12}^2}+\frac{3}{4}\left(\ln\frac{\mu_R^2}{Q_{13}^2}
+\ln\frac{\mu_R^2}{Q_{23}^2}\right)\right)\\[0.5ex]
&\qquad+C_F\left(-4
+\operatorname{Li}_2\left(-\frac{Q_{13}^2}{M^2}\right)
+\operatorname{Li}_2\left(-\frac{Q_{23}^2}{M^2}\right)
+\frac{3}{2}\ln\frac{\mu_R^2}{Q_{12}^2}+\frac{1}{2}\ln^2\frac{Q_{12}^2}{\mu_R^2}
+\frac{1}{2}\ln^2\frac{Q_{13}^2}{Q_{23}^2}\right.\\[0.5ex]
&\qquad\left.-\frac{1}{2}\ln^2\frac{Q_{13}^2+M^2}{Q_{13}^2}
-\frac{1}{2}\ln^2\frac{Q_{23}^2+M^2}{Q_{23}^2}+\frac{1}{2}\ln^2\frac{M^2+Q_{13}^2}{M^2}
+\frac{1}{2}\ln^2\frac{M^2+Q_{23}^2}{M^2}\right)\,.
\end{aligned}
\end{equation}

The two hard functions therefore differ only by terms fixed by the colour
charges and directions of the external legs, which are unchanged for
$q\bar q\to Zg$. The relation obtained here can then be used in
Sec.~\ref{subsec:NNLL-coeff} to compare the \textsc{ARES} and SCET expansion
coefficients.
