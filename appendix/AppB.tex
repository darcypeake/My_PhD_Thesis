\chapter{Additive observables}
\label{ap:add-res-NNLL}

For additive observables (cf.\ Sec.~\ref{subsec:additive}), the NNLL $\mathcal{F}$-functions
entering $\delta\mathcal{F}_{\rm NNLL}$, defined in Eq.~\eqref{eq:dFNNLL}, take the general
form derived in~\cite{Arpino:2019ozn}. These contributions are expressed in terms of
$\lambda \equiv \alpha_s(Q)\beta_0\ln(1/v)$ and the NLL-function $\FNLL(\lambda)$, whose additive form is given
in Eq.~\eqref{eq:FNLL-add-final}. 
 
The soft-collinear correction is
\begin{multline}
 \label{eq:dFsc-general-add}
\delta\mathcal{F}_{\mathrm{sc}}(\lambda)=-\FNLL(\lambda)\frac{\pi}{\alpha_s(Q)}\sum_{\ell=1}^3\left[\left(\Delta R'_{\ell,\mathrm{NNLL}}+ R''_{\ell,\mathrm{NNLL}}\left(\langle \ln d_\ell g_\ell(\phi)\rangle-b_\ell\ln\frac{2E_\ell}{Q}\right)\right)\right. \\
\times \left.\left(\psi^{(0)}\left(1+\RpNLL\right)+\gamma_E\right)+\frac{R''_{\ell,\mathrm{NNLL}}}{2}\left(\left(\psi^{(0)}(1+\RpNLL)+\gamma_E\right)^2-\psi^{(1)}(1+\RpNLL)+\frac{\pi^2}{6}\right)\right]\,,
\end{multline}
where $\RpNLL, R''_{\ell,\mathrm{NNLL}}$, and $\Delta R'_{\ell,\mathrm{NNLL}}$
are defined in Eqs.~\eqref{eq:RpNLL}, \eqref{eq:R2NNLL}, and~\eqref{eq:DRNNLL} respectively. 

The correlated correction is given by
\begin{equation}
\begin{aligned}
\label{eq:dFcorrel-general-add}
&\delta\mathcal{F}_{\mathrm{correl}}(\lambda)=-\FNLL(\lambda)
\sum_{\ell=1}^3\frac{\lambda R''_{\ell,\mathrm{NNLL}}}{2a\beta_0\alpha_s(Q)}\\
&\qquad\times
\int_0^{2\pi}\frac{d\phi^{(\ell)}}{2\pi}\int_0^\infty
\frac{d\mu^2}{\mu^2(1+\mu^2)}\int_0^1dz\int_0^{2\pi}\frac{d\phi}{2\pi}\frac{1}{2!}\mathcal{A}^2(z,\mu,\phi)
\ln f^{(\ell)}_{\mathrm{correl}}\left(z,\mu,\phi,\phi^{(\ell)}\right)\,,
\end{aligned}
\end{equation}
where
\begin{equation}
	\frac{\Vsc{k_a,k_b,\{k_i\}}}{v}=\zeta f_{\rm correl}(z,\mu,\phi,\phi^{(\ell)})+\sum_{i=1}^n\zeta_i\,.
\end{equation}

The hard-collinear contribution is
\begin{equation}
  \label{eq:dFhc-general-add}
   \delta\mathcal{F}_{\rm hc}=-\FNLL(\lambda)\left(\psi^{(0)}(1+\RpNLL)+\gamma_E\right)\sum_\ell \frac{\alpha_s\left(Qv^{1/(a+b_\ell)}\right)}{\alpha_s(Q)\left(a+b_\ell\right)}B_\ell \,,
\end{equation}
where $B_\ell=\gamma_\ell^{(0)}$ for final-state legs, while for
initial-state legs it is defined in Eq.~\eqref{eq:Bell-ISR}.

The recoil correction $\delta\mathcal{F}_{\rm rec}$ is
\begin{equation}
  \label{eq:dFrec-general-add}
  \delta\mathcal{F}_{\rm rec} = \FNLL(\lambda)
  \sum_\ell 
  \frac{\alpha_s\!\left(Qv^{1/(a+b_\ell)}\right)}
       {\alpha_s(Q)(a+b_\ell)}
 \int_0^{2\pi} \frac{d\phi^{(\ell)}}{2\pi}\int_0^1 dz^{(\ell)}\,P_\ell(z)\ln \frac{f_{\rm sc}^{(\ell)}(z^{(\ell)},\phi^{(\ell)})}{f_{\rm hc}^{(\ell)}(z^{(\ell)},\phi^{(\ell)})}
\end{equation}
where the function $P_{\ell}(z)$ is given in Eqs.~\eqref{eq:Pell-ISR} and~\eqref{eq:Pell-FSR}
for initial- and final-state radiation, respectively. The observable-dependent functions are defined
through
\begin{subequations}
\begin{align}
&\frac{V_{\rm hc}(\{\tilde{p}\},k,\{k_i\})}{v}=\zeta f_{\rm hc}^{(\ell)}(z^{(\ell)},\phi^{(\ell)})+\sum_i\zeta_i\,,\\
&\frac{V_{\rm sc}(\{\tilde{p}\},k,\{k_i\})}{v}=\zeta f_{\rm sc}^{(\ell)}(z^{(\ell)},\phi^{(\ell)})+\sum_i\zeta_i\,.
\end{align}
\end{subequations}

The recoil correction $\Delta\mathcal{F}_{\rm rec}$ is
\begin{multline}
 \label{eq:DFrec-general-add}
	\Delta\mathcal{F}_{\rm rec}(\lambda)=-\FNLL(\lambda)\sum_{\ell_g}\frac{\alpha_s(Qv^{1/(a+b_{\ell_g})})}{\alpha_s(Q)(a+b_{\ell_g})}\\\times\int_0^{\pi}\frac{d\phi^{(\ell_g)}}{\pi}\left(2\cos^2\phi^{(\ell_g)}-1\right)\int_0^1dz^{(\ell_g)}\, \Delta P_g(z^{(\ell_g)})\ln f_{\rm hc}^{(\ell_g)}(z^{(\ell_g)},\phi^{(\ell_g)})\,,
\end{multline}
where the sum runs over the gluon legs $\ell_g$ and
$\Delta P_g(z)$ are given by
Eq.~\eqref{eq:DPg-ISR} for initial-state radiation and
Eq.~\eqref{eq:DPg-FSR} for final-state radiation.

The wide-angle correction $\delta\mathcal{F}_{\rm wa}(\lambda)$ is
\begin{equation}
 \label{eq:dFwa-general-add}
	\delta\mathcal{F}_{\rm wa}(\lambda)=\FNLL(\lambda)
	\sum_{(\ell\bar\ell)}\frac{2C_{\ell\bar\ell}}{a}\frac{\alpha_s(v^{1/a}Q)}{\alpha_s(Q)}\int_{-\infty}^{\infty}d\eta^{(\ell\bar\ell)}\int_0^{2\pi}\frac{d\phi^{(\ell\bar\ell)}}{2\pi}\ln\frac{f_{\rm sc}^{(\ell\bar\ell)}(\eta^{(\ell\bar\ell)},\phi^{(\ell\bar\ell)})}{f_{\rm wa}^{(\ell\bar\ell)}(\eta^{(\ell\bar\ell)},\phi^{(\ell\bar\ell)})}\,,
\end{equation}
where
\begin{equation}
	\frac{V_{\rm wa}(\{\tilde{p}\},k^{(\ell\bar\ell)},\{k_i\})}{v}=\zeta f_{\rm wa}^{(\ell\bar\ell)}(\eta^{(\ell\bar\ell)},\phi^{(\ell\bar\ell)})+\sum_i\zeta_i\,.
\end{equation}

Finally, the wide-angle correction $\Delta\mathcal{F}_{\rm wa}(\lambda)$ takes the form
 \begin{equation}
 \label{eq:DFwa-general-add}
 	\Delta\mathcal{F}_{\mathrm{wa}}(\lambda)=-\FNLL(\lambda)\left(\psi^{(0)}\left(1+R'_{\mathrm{NLL}}\right)+\gamma_E\right)\frac{\alpha_s\left(v^{1/a}Q\right)}{a\alpha_s(Q)}2\sum_{(\ell\bar\ell)}C_{\ell\bar\ell}\ln\left(\frac{Q_{\ell\bar\ell}}{Q}\right)\,.
 \end{equation}
